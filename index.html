// อัปเดตฟังก์ชันลงทะเบียนรถ เพื่อรองรับข้อมูลชุดใหม่ทั้งหมด function addVehicle(data) { const lock = LockService.getScriptLock(); if (lock.tryLock(10000)) { try { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); // 1. อัปเดตข้อมูลผู้ใช้ (ชื่อ, เบอร์, เพศ, อายุ) ลงชีต Users const userSheet = ss.getSheetByName('Users'); const usersData = userSheet.getDataRange().getValues(); for (let i = 1; i < usersData.length; i++) { if (usersData[i][1] === data.lineUid) { userSheet.getRange(i + 1, 3).setValue(data.uName || usersData[i][2]); // อัปเดตชื่อ userSheet.getRange(i + 1, 4).setValue(data.uPhone || usersData[i][3]); // อัปเดตเบอร์ // สมมติว่าคอลัมน์ H (8) คือ Gender, คอลัมน์ I (9) คือ Age userSheet.getRange(i + 1, 8).setValue(data.uGender || ''); userSheet.getRange(i + 1, 9).setValue(data.uAge || ''); break; } } // 2. บันทึกข้อมูลรถยนต์ใหม่ลงชีต Vehicles ss.getSheetByName('Vehicles').appendRow([ 'VEH-' + Date.now(), data.lineUid, data.licensePlate, data.province || '', // คอลัมน์ใหม่: จังหวัด data.brand || '', // คอลัมน์ใหม่: ยี่ห้อ data.model, data.color, data.size || 'M', // คอลัมน์ใหม่: ขนาด (ถ้าไม่ระบุให้เป็น M) new Date() ]); SpreadsheetApp.flush(); return { status: 'success' }; } catch (e) { return { status: 'error', message: e.message }; } finally { lock.releaseLock(); } } else { return { status: 'error', message: 'ระบบไม่ว่าง' }; } }
